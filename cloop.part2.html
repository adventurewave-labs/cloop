//! Loop configuration: the data model, a tiny TOML-subset (de)serializer,
//! Claude Code argument construction, and per-iteration prompt rendering.
//!
//! The serializer deliberately supports only a flat subset of TOML — top-level
//! `key = value` pairs, basic strings, integers, booleans, and triple-quoted
//! multi-line strings. That keeps the parser ~80 lines and dependency-free.
//! Assumption: a multi-line value never contains a line that is exactly `"""`.

use crate::ui;
use std::collections::BTreeMap;

/// How a loop decides it is finished.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StopKind {
    /// Run a fixed number of iterations.
    Iterations,
    /// Stop when Claude prints a marker string.
    Marker,
    /// Stop when a shell command exits 0.
    Command,
}

impl StopKind {
    pub fn as_str(self) -> &'static str {
        match self {
            StopKind::Iterations => "iterations",
            StopKind::Marker => "marker",
            StopKind::Command => "command",
        }
    }
    pub fn from_str(s: &str) -> StopKind {
        match s {
            "iterations" => StopKind::Iterations,
            "marker" => StopKind::Marker,
            _ => StopKind::Command,
        }
    }
}

/// A single saved loop.
#[derive(Debug, Clone)]
pub struct LoopConfig {
    pub name: String,
    pub task: String,
    pub workdir: String,
    pub model: String,
    pub stop_kind: StopKind,
    pub stop_command: String,
    pub stop_marker: String,
    pub max_iterations: u64,
    pub carry_context: bool,
    pub permission_mode: String,
    pub skip_permissions: bool,
    pub max_turns: u64,
    pub allowed_tools: String,
    pub delay_secs: u64,
    pub extra_args: String,
    pub iter_prompt: String,
}
