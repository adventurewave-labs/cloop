<span class="com">/// Turn arbitrary text into a safe lowercase filename stem.</span>
<span class="kw">fn</span> sanitize_name(s: &amp;<span class="ty">str</span>) -&gt; <span class="ty">String</span> {
    <span class="kw">let</span> <span class="kw">mut</span> out = <span class="ty">String</span>::new();
    <span class="kw">for</span> ch <span class="kw">in</span> s.trim().chars() {
        <span class="kw">if</span> ch.is_ascii_alphanumeric() {
            out.push(ch.to_ascii_lowercase());
        } <span class="kw">else</span> <span class="kw">if</span> ch == <span class="chr">'-'</span> || ch == <span class="chr">'_'</span> {
            out.push(ch);
        } <span class="kw">else</span> <span class="kw">if</span> ch.is_whitespace() {
            out.push(<span class="chr">'-'</span>);
        }
        <span class="com">// anything else is dropped</span>
    }
    <span class="kw">let</span> trimmed = out.trim_matches(<span class="chr">'-'</span>).to_string();
    <span class="kw">if</span> trimmed.is_empty() {
        <span class="str">"loop"</span>.to_string()
    } <span class="kw">else</span> {
        trimmed
    }
}

<span class="kw">fn</span> indent(s: &amp;<span class="ty">str</span>, pad: &amp;<span class="ty">str</span>) -&gt; <span class="ty">String</span> {
    s.lines()
        .map(|l| <span class="mac">format!</span>(<span class="str">"{}{}"</span>, pad, l))
        .collect::&lt;<span class="ty">Vec</span>&lt;_&gt;&gt;()
        .join(<span class="str">"\n"</span>)
}

    <span class="mac">println!</span>();
    <span class="kw">let</span> stops = [

        <span class="str">"iterations — run a fixed number of times"</span>,

    <span class="kw">let</span> s = ui::ask_select(<span class="str">"Stop condition"</span>, &amp;stops, <span class="num">0</span>)?;

        StopKind::Iterations =&gt; {}

        <span class="num">2</span> =&gt; cfg.permission_mode = <span class="str">"plan"</span>.into(),

    };

    <span class="ty">Ok</span>(cfg)
