<span class="com">/// Split a string into arguments, honoring single and double quotes. Quotes</span>
<span class="com">/// group tokens but are not themselves included in the output.</span>
<span class="kw">pub</span> <span class="kw">fn</span> split_args(s: &amp;<span class="ty">str</span>) -&gt; <span class="ty">Vec</span>&lt;<span class="ty">String</span>&gt; {
    <span class="kw">let</span> <span class="kw">mut</span> out: <span class="ty">Vec</span>&lt;<span class="ty">String</span>&gt; = <span class="ty">Vec</span>::new();
    <span class="kw">let</span> <span class="kw">mut</span> cur = <span class="ty">String</span>::new();
    <span class="kw">let</span> <span class="kw">mut</span> in_single = <span class="kw">false</span>;
    <span class="kw">let</span> <span class="kw">mut</span> in_double = <span class="kw">false</span>;
    <span class="kw">let</span> <span class="kw">mut</span> has_token = <span class="kw">false</span>;
    <span class="kw">for</span> ch <span class="kw">in</span> s.chars() {
        <span class="kw">if</span> in_single {
            <span class="kw">if</span> ch == <span class="chr">'\''</span> {
                in_single = <span class="kw">false</span>;
            } <span class="kw">else</span> {
                cur.push(ch);
            }
        } <span class="kw">else</span> <span class="kw">if</span> in_double {
            <span class="kw">if</span> ch == <span class="chr">'"'</span> {
                in_double = <span class="kw">false</span>;
            } <span class="kw">else</span> {
                cur.push(ch);
            }
        } <span class="kw">else</span> <span class="kw">if</span> ch == <span class="chr">'\''</span> {
            in_single = <span class="kw">true</span>;
            has_token = <span class="kw">true</span>;
        } <span class="kw">else</span> <span class="kw">if</span> ch == <span class="chr">'"'</span> {
            in_double = <span class="kw">true</span>;
            has_token = <span class="kw">true</span>;
        } <span class="kw">else</span> <span class="kw">if</span> ch.is_whitespace() {
            <span class="kw">if</span> has_token {
                out.push(std::mem::take(&amp;<span class="kw">mut</span> cur));
                has_token = <span class="kw">false</span>;
            }
        } <span class="kw">else</span> {
            cur.push(ch);
            has_token = <span class="kw">true</span>;
        }
    }
    <span class="kw">if</span> has_token {
        out.push(cur);
    }
    out
}

<span class="kw">fn</span> esc_basic(s: &amp;<span class="ty">str</span>) -&gt; <span class="ty">String</span> {
    <span class="kw">let</span> <span class="kw">mut</span> out = <span class="ty">String</span>::with_capacity(s.len() + <span class="num">2</span>);
    <span class="kw">for</span> ch <span class="kw">in</span> s.chars() {
        <span class="kw">match</span> ch {
            <span class="chr">'\\'</span> =&gt; out.push_str(<span class="str">"\\\\"</span>),
            <span class="chr">'"'</span> =&gt; out.push_str(<span class="str">"\\\""</span>),
            <span class="chr">'\n'</span> =&gt; out.push_str(<span class="str">"\\n"</span>),
            <span class="chr">'\t'</span> =&gt; out.push_str(<span class="str">"\\t"</span>),
            <span class="chr">'\r'</span> =&gt; out.push_str(<span class="str">"\\r"</span>),
            _ =&gt; out.push(ch),
        }
    }
    out
}

<span class="kw">fn</span> unescape_basic(s: &amp;<span class="ty">str</span>) -&gt; <span class="ty">String</span> {
    <span class="kw">let</span> <span class="kw">mut</span> out = <span class="ty">String</span>::with_capacity(s.len());
    <span class="kw">let</span> <span class="kw">mut</span> chars = s.chars();
    <span class="kw">while</span> <span class="kw">let</span> <span class="ty">Some</span>(ch) = chars.next() {
        <span class="kw">if</span> ch == <span class="chr">'\\'</span> {
            <span class="kw">match</span> chars.next() {
                <span class="ty">Some</span>(<span class="chr">'\\'</span>) =&gt; out.push(<span class="chr">'\\'</span>),
                <span class="ty">Some</span>(<span class="chr">'"'</span>) =&gt; out.push(<span class="chr">'"'</span>),
                <span class="ty">Some</span>(<span class="chr">'n'</span>) =&gt; out.push(<span class="chr">'\n'</span>),
                <span class="ty">Some</span>(<span class="chr">'t'</span>) =&gt; out.push(<span class="chr">'\t'</span>),
                <span class="ty">Some</span>(<span class="chr">'r'</span>) =&gt; out.push(<span class="chr">'\r'</span>),
                <span class="ty">Some</span>(other) =&gt; {
                    out.push(<span class="chr">'\\'</span>);
                    out.push(other);
                }
                <span class="ty">None</span> =&gt; out.push(<span class="chr">'\\'</span>),
            }
        } <span class="kw">else</span> {
            out.push(ch);
        }
    }
    out
}

<span class="kw">fn</span> emit_multiline(out: &amp;<span class="kw">mut</span> <span class="ty">String</span>, key: &amp;<span class="ty">str</span>, val: &amp;<span class="ty">str</span>) {
    out.push_str(key);
    out.push_str(<span class="str">" = \"\"\"\n"</span>);
    out.push_str(val);
    out.push(<span class="chr">'\n'</span>);
    out.push_str(<span class="str">"\"\"\"\n"</span>);
}

<span class="com">/// Parse our TOML subset. Unknown keys are ignored; missing keys use defaults.</span>
<span class="kw">pub</span> <span class="kw">fn</span> from_toml(text: &amp;<span class="ty">str</span>) -&gt; <span class="ty">Result</span>&lt;LoopConfig, <span class="ty">String</span>&gt; {
    <span class="kw">let</span> <span class="kw">mut</span> map: <span class="ty">BTreeMap</span>&lt;<span class="ty">String</span>, <span class="ty">String</span>&gt; = <span class="ty">BTreeMap</span>::new();
    <span class="kw">let</span> <span class="kw">mut</span> lines = text.lines();

        }
