    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"name"</span>) {

    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"task"</span>) {
        cfg.task = v.clone();
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"workdir"</span>) {
        <span class="kw">if</span> !v.is_empty() {
            cfg.workdir = v.clone();
        }
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"model"</span>) {
        cfg.model = v.clone();
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"stop_kind"</span>) {
        cfg.stop_kind = StopKind::from_str(v);
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"stop_command"</span>) {
        cfg.stop_command = v.clone();
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"stop_marker"</span>) {
        <span class="kw">if</span> !v.is_empty() {
            cfg.stop_marker = v.clone();
        }
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"max_iterations"</span>) {
        <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Ok</span>(n) = v.trim().parse::&lt;<span class="ty">u64</span>&gt;() {

        }
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"carry_context"</span>) {

    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"permission_mode"</span>) {
        cfg.permission_mode = v.clone();
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"skip_permissions"</span>) {
        cfg.skip_permissions = parse_bool(v, cfg.skip_permissions);
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"max_turns"</span>) {
        <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Ok</span>(n) = v.trim().parse::&lt;<span class="ty">u64</span>&gt;() {

        }
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"allowed_tools"</span>) {

    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"delay_secs"</span>) {

        }
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"extra_args"</span>) {
        cfg.extra_args = v.clone();
    }
    <span class="kw">if</span> <span class="kw">let</span> <span class="ty">Some</span>(v) = map.get(<span class="str">"iter_prompt"</span>) {
        cfg.iter_prompt = v.clone();
    }
    <span class="ty">Ok</span>(cfg)
}

<span class="kw">fn</span> parse_bool(s: &amp;<span class="ty">str</span>, default: <span class="ty">bool</span>) -&gt; <span class="ty">bool</span> {
    <span class="kw">match</span> s.trim() {
        <span class="str">"true"</span> =&gt; <span class="kw">true</span>,
        <span class="str">"false"</span> =&gt; <span class="kw">false</span>,
        _ =&gt; default,
    }
}

<span class="com">/// A human-readable, colored summary block (shown before saving / running).</span>
<span class="kw">pub</span> <span class="kw">fn</span> summary(cfg: &amp;LoopConfig) -&gt; <span class="ty">String</span> {
    <span class="kw">let</span> <span class="kw">mut</span> rows: <span class="ty">Vec</span>&lt;(<span class="ty">String</span>, <span class="ty">String</span>)&gt; = <span class="ty">Vec</span>::new();
    rows.push((<span class="str">"name"</span>.into(), cfg.name.clone()));

    <span class="kw">let</span> task_first = cfg.task.lines().next().unwrap_or(<span class="str">""</span>).to_string();
    <span class="kw">let</span> task_disp = <span class="kw">if</span> cfg.task.lines().count() &gt; <span class="num">1</span> {
        <span class="mac">format!</span>(<span class="str">"{} {}"</span>, task_first, ui::dim(<span class="str">"(…)"</span>))
    } <span class="kw">else</span> {
        task_first
    };
    rows.push((<span class="str">"task"</span>.into(), task_disp));
    rows.push((<span class="str">"workdir"</span>.into(), cfg.workdir.clone()));
    rows.push((
        <span class="str">"model"</span>.into(),
        <span class="kw">if</span> cfg.model.is_empty() {
            <span class="str">"default"</span>.into()
        } <span class="kw">else</span> {
            cfg.model.clone()
        },
    ));
    <span class="kw">let</span> stop = <span class="kw">match</span> cfg.stop_kind {
        StopKind::<span class="ty">Command</span> =&gt; <span class="mac">format!</span>(<span class="str">"command → {}"</span>, cfg.stop_command),
        StopKind::Marker =&gt; <span class="mac">format!</span>(<span class="str">"marker → {}"</span>, cfg.stop_marker),
        StopKind::Iterations =&gt; <span class="str">"fixed iterations"</span>.to_string(),
    };
    rows.push((<span class="str">"stop"</span>.into(), stop));
    rows.push((<span class="str">"max iters"</span>.into(), cfg.max_iterations.to_string()));
    rows.push((
        <span class="str">"carry context"</span>.into(),
        <span class="kw">if</span> cfg.carry_context { <span class="str">"yes"</span> } <span class="kw">else</span> { <span class="str">"no"</span> }.into(),
    ));
    <span class="kw">let</span> perms = <span class="kw">if</span> cfg.skip_permissions {
        ui::red(<span class="str">"bypass (--dangerously-skip-permissions)"</span>)
    } <span class="kw">else</span> <span class="kw">if</span> cfg.permission_mode.is_empty() {
        <span class="str">"default"</span>.to_string()
    } <span class="kw">else</span> {
        cfg.permission_mode.clone()
    };
    rows.push((<span class="str">"permissions"</span>.into(), perms));
    <span class="kw">if</span> cfg.max_turns &gt; <span class="num">0</span> {
        rows.push((<span class="str">"max turns"</span>.into(), cfg.max_turns.to_string()));
    }
    <span class="kw">if</span> !cfg.allowed_tools.is_empty() {
        rows.push((<span class="str">"allowed tools"</span>.into(), cfg.allowed_tools.clone()));
    }
    <span class="kw">if</span> cfg.delay_secs &gt; <span class="num">0</span> {
        rows.push((<span class="str">"delay"</span>.into(), <span class="mac">format!</span>(<span class="str">"{}s"</span>, cfg.delay_secs)));
    }
    <span class="kw">if</span> !cfg.extra_args.is_empty() {
        rows.push((<span class="str">"extra args"</span>.into(), cfg.extra_args.clone()));
    }

    <span class="kw">let</span> width = rows.iter().map(|(k, _)| k.chars().count()).max().unwrap_or(<span class="num">0</span>);
    <span class="kw">let</span> <span class="kw">mut</span> out = <span class="ty">String</span>::new();
    <span class="kw">for</span> (k, v) <span class="kw">in</span> rows {
        <span class="kw">let</span> <span class="kw">mut</span> key = k;
        <span class="kw">while</span> key.chars().count() &lt; width {
            key.push(<span class="chr">' '</span>);
        }
        out.push_str(&amp;<span class="mac">format!</span>(<span class="str">"  {}  {}\n"</span>, ui::dim(&amp;key), v));
    }
    out
}
