<span class="com">/// Print, without executing, the commands a run would issue.</span>
<span class="kw">pub</span> <span class="kw">fn</span> preview_command(cfg: &amp;LoopConfig) {
    ui::step(<span class="str">"Dry run — commands that would execute:"</span>);
    <span class="mac">println!</span>();

    <span class="kw">let</span> p1 = config::render_prompt(cfg, <span class="num">1</span>, <span class="str">""</span>);
    <span class="kw">let</span> a1 = config::claude_args(cfg, <span class="num">1</span>, &amp;p1);
    <span class="mac">println!</span>(<span class="str">"{}"</span>, ui::bold(<span class="str">"iteration 1:"</span>));
    <span class="mac">println!</span>(<span class="str">"  {}"</span>, display_cmd(&amp;a1));
    <span class="mac">println!</span>();

    <span class="kw">let</span> p2 = config::render_prompt(cfg, <span class="num">2</span>, <span class="str">"&lt;last check output&gt;"</span>);
    <span class="kw">let</span> a2 = config::claude_args(cfg, <span class="num">2</span>, &amp;p2);
    <span class="mac">println!</span>(<span class="str">"{}"</span>, ui::bold(<span class="str">"iteration 2+:"</span>));
    <span class="mac">println!</span>(<span class="str">"  {}"</span>, display_cmd(&amp;a2));

    <span class="kw">if</span> cfg.stop_kind == StopKind::<span class="ty">Command</span> {
        <span class="mac">println!</span>();
        <span class="mac">println!</span>(<span class="str">"{}"</span>, ui::bold(<span class="str">"stop check (each iteration):"</span>));
        <span class="mac">println!</span>(<span class="str">"  sh -c {}"</span>, shell_quote(&amp;cfg.stop_command));
    }
    <span class="mac">println!</span>();
}
