<span class="com">//!</span>
<span class="com">//! The in-Rust runner is the primary path; this is a convenience for CI or for</span>
<span class="com">//! sharing a loop where installing `cloop` isn't worth it. Complex permission</span>
<span class="com">//! or tool specs may need manual quoting in the generated FLAGS line.</span>

<span class="kw">use</span> <span class="kw">crate</span>::config::{LoopConfig, StopKind};

<span class="com">/// Single-quote a string for safe embedding in bash.</span>
<span class="kw">fn</span> sh(s: &amp;<span class="ty">str</span>) -&gt; <span class="ty">String</span> {
    <span class="mac">format!</span>(<span class="str">"'{}'"</span>, s.replace(<span class="chr">'\''</span>, <span class="str">"'\\''"</span>))
}

<span class="com">/// Produce a bash script that approximates running this loop.</span>
<span class="kw">pub</span> <span class="kw">fn</span> to_bash(cfg: &amp;LoopConfig) -&gt; <span class="ty">String</span> {
    <span class="kw">let</span> <span class="kw">mut</span> s = <span class="ty">String</span>::new();
    s.push_str(<span class="str">"#!/usr/bin/env bash\n"</span>);
    s.push_str(<span class="str">"# Generated by cloop — an equivalent loop as a portable shell script.\n"</span>);
    s.push_str(<span class="str">"set -uo pipefail\n\n"</span>);

    s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"cd {} || exit 1\n\n"</span>, sh(&amp;cfg.workdir)));

    s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"MAX={}\n"</span>, cfg.max_iterations));
    s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"TASK={}\n"</span>, sh(&amp;cfg.task)));
    <span class="kw">match</span> cfg.stop_kind {
        StopKind::<span class="ty">Command</span> =&gt; s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"CHECK={}\n"</span>, sh(&amp;cfg.stop_command))),
        StopKind::Marker =&gt; s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"MARKER={}\n"</span>, sh(&amp;cfg.stop_marker))),
        StopKind::Iterations =&gt; {}
    }
    s.push(<span class="chr">'\n'</span>);

    <span class="com">// Static flags (everything except --continue and the positional prompt).</span>
    <span class="kw">let</span> <span class="kw">mut</span> flags: <span class="ty">Vec</span>&lt;<span class="ty">String</span>&gt; = <span class="mac">vec!</span>[<span class="str">"--print"</span>.to_string()];
    <span class="kw">if</span> !cfg.model.is_empty() {
        flags.push(<span class="str">"--model"</span>.into());
        flags.push(cfg.model.clone());
    }
    <span class="kw">if</span> cfg.skip_permissions {
        flags.push(<span class="str">"--dangerously-skip-permissions"</span>.into());
    } <span class="kw">else</span> <span class="kw">if</span> !cfg.permission_mode.is_empty() {
        flags.push(<span class="str">"--permission-mode"</span>.into());
        flags.push(cfg.permission_mode.clone());
    }
    <span class="kw">if</span> cfg.max_turns &gt; <span class="num">0</span> {
        flags.push(<span class="str">"--max-turns"</span>.into());
        flags.push(cfg.max_turns.to_string());
    }
    <span class="kw">if</span> !cfg.allowed_tools.is_empty() {
        flags.push(<span class="str">"--allowedTools"</span>.into());
        flags.push(cfg.allowed_tools.clone());
    }
    <span class="kw">if</span> !cfg.extra_args.is_empty() {
        flags.push(cfg.extra_args.clone());
    }
    s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"FLAGS=\"{}\"\n\n"</span>, flags.join(<span class="str">" "</span>)));

    s.push_str(<span class="str">"for ((i=1; i&lt;=MAX; i++)); do\n"</span>);
    s.push_str(<span class="str">"  echo \"── iteration $i / $MAX ──\"\n"</span>);
    <span class="kw">if</span> cfg.carry_context {
        s.push_str(<span class="str">"  cont=\"\"; [ \"$i\" -gt 1 ] &amp;&amp; cont=\"--continue\"\n"</span>);
    } <span class="kw">else</span> {
        s.push_str(<span class="str">"  cont=\"\"\n"</span>);
    }

    <span class="kw">match</span> cfg.stop_kind {
        StopKind::<span class="ty">Command</span> =&gt; {
            s.push_str(<span class="str">"  if [ \"$i\" -eq 1 ]; then\n"</span>);
            s.push_str(<span class="str">"    prompt=\"$TASK\"\n"</span>);
            s.push_str(<span class="str">"  else\n"</span>);
            s.push_str(<span class="str">"    prompt=\"Iteration $i of $MAX. The check '$CHECK' is still failing. Its output was:\n\n$out\n\nDiagnose and fix it with concrete edits.\"\n"</span>);
            s.push_str(<span class="str">"  fi\n"</span>);
            s.push_str(<span class="str">"  claude $FLAGS $cont \"$prompt\"\n"</span>);
            s.push_str(<span class="str">"  if out=\"$(eval \"$CHECK\" 2&gt;&amp;1)\"; then\n"</span>);
            s.push_str(<span class="str">"    echo \"check passed on iteration $i\"; exit 0\n"</span>);
            s.push_str(<span class="str">"  fi\n"</span>);
        }
        StopKind::Marker =&gt; {
            s.push_str(<span class="str">"  if [ \"$i\" -eq 1 ]; then\n"</span>);
            s.push_str(<span class="str">"    prompt=\"$TASK\n\nWhen fully complete, print the exact line $MARKER on its own line.\"\n"</span>);
            s.push_str(<span class="str">"  else\n"</span>);
            s.push_str(<span class="str">"    prompt=\"Iteration $i of $MAX. Continue. When fully complete, print the exact line $MARKER on its own line.\"\n"</span>);
            s.push_str(<span class="str">"  fi\n"</span>);
            s.push_str(<span class="str">"  output=\"$(claude $FLAGS $cont \"$prompt\")\"\n"</span>);
            s.push_str(<span class="str">"  echo \"$output\"\n"</span>);
            s.push_str(<span class="str">"  if printf '%s' \"$output\" | grep -qF \"$MARKER\"; then\n"</span>);
            s.push_str(<span class="str">"    echo \"marker seen on iteration $i\"; exit 0\n"</span>);
            s.push_str(<span class="str">"  fi\n"</span>);
        }
        StopKind::Iterations =&gt; {
            s.push_str(<span class="str">"  if [ \"$i\" -eq 1 ]; then\n"</span>);
            s.push_str(<span class="str">"    prompt=\"$TASK\"\n"</span>);
            s.push_str(<span class="str">"  else\n"</span>);
            s.push_str(<span class="str">"    prompt=\"Iteration $i of $MAX. Continue working on the task.\"\n"</span>);
            s.push_str(<span class="str">"  fi\n"</span>);
            s.push_str(<span class="str">"  claude $FLAGS $cont \"$prompt\"\n"</span>);
        }
    }

    <span class="kw">if</span> cfg.delay_secs &gt; <span class="num">0</span> {
        s.push_str(&amp;<span class="mac">format!</span>(<span class="str">"  sleep {}\n"</span>, cfg.delay_secs));
    }
    s.push_str(<span class="str">"done\n\n"</span>);

    <span class="kw">match</span> cfg.stop_kind {
        StopKind::Iterations =&gt; s.push_str(<span class="str">"echo \"completed $MAX iterations\"\n"</span>),
        _ =&gt; {
            s.push_str(<span class="str">"echo \"reached MAX without satisfying the stop condition\" &gt;&amp;2\n"</span>);
            s.push_str(<span class="str">"exit 1\n"</span>);
        }
    }

    s
